# Multiprocessing serialisation — replaces pickle for lambdas and closures
dill>=0.3

# GPU support — choose the variant matching your CUDA driver (nvidia-smi):
#   CUDA 12.x  →  cupy-cuda12x   (Blackwell, Ada, Ampere, most modern GPUs)
#   CUDA 11.x  →  cupy-cuda11x   (Turing, Volta)
cupy-cuda12x>=12.0

# CUDA 12 runtime shared libraries for cupy-cuda12x.
#
# WHY: pip install -e . installs torch, which pulls in nvidia-cublas-13.*
# (a CUDA 13 build).  cupy-cuda12x requires libcublas.so.12, not .so.13.
# Installing the -cu12 wheels below makes both coexist correctly.
#
# After installing, run:
#   python scripts/set_ld_library_path.py
# to register the .so paths so CuPy can find them at runtime.
nvidia-cublas-cu12
nvidia-cufft-cu12
nvidia-curand-cu12
nvidia-cusolver-cu12
nvidia-cusparse-cu12
nvidia-cuda-runtime-cu12
nvidia-cuda-nvrtc-cu12
